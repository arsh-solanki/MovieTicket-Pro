JAVASCRIPT EXECUTION:
  $('#reserveButton').click(function() {
      // 1. Build request object
      const request = {
          showId: 15,
          seatNumbers: ["A1", "A2", "B3"]
      };

      // 2. Get antiforgery token from hidden input
      const token = $('input[name="__RequestVerificationToken"]').val();
      // token = "CfDJ8..." (auto-generated by @Html.AntiForgeryToken())

      // 3. Perform AJAX POST with JSON
      $.ajax({
          url: '/Booking/ReserveSeats',          // Server endpoint
          type: 'POST',                          // HTTP method
          contentType: 'application/json; charset=utf-8',  // Request content type
          dataType: 'json',                      // Expected response format
          headers: {
              'RequestVerificationToken': token  // CSRF protection
          },
          data: JSON.stringify(request),          // Serialize request to JSON
          success: function(response) {
              // Handle success response
              if (response.success) {
                  // Redirect to confirmation page
                  window.location.href = '/Booking/Confirmation/' + response.bookingId;
              } else {
                  // Show error message
                  alert('Booking failed: ' + response.message);
              }
          },
          error: function(xhr, status, error) {
              // Handle network error or 500 response
              alert('Error: ' + error);
          }
      });
  });

SERVER PROCESSING (BookingController.ReserveSeats):
  
  Entry:
    [HttpPost("ReserveSeats")]
    [Authorize]                    // Requires logged-in user
    [ValidateAntiForgeryToken]     // Validates token from header
    public async Task<IActionResult> ReserveSeats([FromBody] ReserveSeatsRequest request)

  Step 1: Validate request format
    ├─ Check: request != null
    ├─ Check: request.ShowId > 0
    ├─ Check: request.SeatNumbers != null && count > 0
    └─ If failed → return Json({ success: false, message: "Invalid request" })

  Step 2: Get authenticated user ID
    ├─ userId = User.FindFirstValue(ClaimTypes.NameIdentifier)
    ├─ Example: userId = "f7c5d1e8-2a4b-4c6d-8e9f-1a2b3c4d5e6f"
    └─ If empty → return Json({ success: false, message: "Authentication required" })

  Step 3: Load show data
    var show = _context.Shows
        .Include(s => s.Movie)
        .Include(s => s.Screen)
        .Include(s => s.SeatPrices)
        .Include(s => s.BookedSeats.Where(bs => bs.IsActive))  // CRITICAL: Only active
        .FirstOrDefaultAsync(s => s.Id == 15 && s.IsActive);

    └─ If not found → return Json({ success: false, message: "Show not found" })

  Step 4: Validate show is available
    if (show.StartDateTime <= DateTime.UtcNow)
        → return Json({ success: false, message: "Show has already started" })

  Step 5: CRITICAL - Check seat availability (RACE CONDITION PREVENTION)
    var alreadyBooked = show.BookedSeats
        .Where(bs => ["A1", "A2", "B3"].Contains(bs.SeatNumber))
        .Select(bs => bs.SeatNumber)
        .ToList();
    // alreadyBooked = [] (empty, none are taken)

    If any seats in alreadyBooked:
        → return Json({
            success: false,
            message: "Some seats are no longer available",
            errorDetails: "Seats A1, A2 have been booked by another user"
        })
    (This prevents race condition where seats were booked after page load)

  Step 6: START DATABASE TRANSACTION
    using (var transaction = await _context.Database.BeginTransactionAsync())
    {
        try
        {
            // All database operations below are atomic:
            // Either ALL succeed and commit, or ALL rollback on any error

            Step 6a: Create Booking record
              var booking = new Booking
              {
                  UserId = "f7c5d1e8-2a4b-4c6d-8e9f-1a2b3c4d5e6f",
                  ShowId = 15,
                  BookingDateTime = DateTime.UtcNow,  // 2026-02-07 10:30:45.1234567Z
                  Status = BookingStatus.Confirmed,
                  IsActive = true,
                  TotalAmount = 0  // Will update after calculating seat prices
              };

              _context.Bookings.Add(booking);
              await _context.SaveChangesAsync();
              // Booking now has Id (auto-generated: 42)

            Step 6b: Create BookedSeat records
              For each seatNumber in ["A1", "A2", "B3"]:

                Iteration 1: "A1"
                  ├─ category = DetermineSeatCategory("A1", layoutJson)
                  │  └─ A1[0] = 'A' → "Premium"
                  ├─ seatPrice = show.SeatPrices.FirstOrDefault(p => p.Category == "Premium")
                  │  └─ seatPrice.Price = 300
                  ├─ Create BookedSeat:
                  │   {
                  │       BookingId: 42,
                  │       ShowId: 15,
                  │       SeatNumber: "A1",
                  │       Category: "Premium",
                  │       Price: 300,
                  │       IsActive: true
                  │   }
                  ├─ _context.BookedSeats.Add(bookedSeat)
                  └─ totalAmount += 300  // totalAmount = 300

                Iteration 2: "A2"
                  ├─ category = "Premium"
                  ├─ price = 300
                  └─ totalAmount += 300  // totalAmount = 600

                Iteration 3: "B3"
                  ├─ category = DetermineSeatCategory("B3", layoutJson)
                  │  └─ B3[0] = 'B' → "Regular"
                  ├─ price = 200
                  └─ totalAmount += 200  // totalAmount = 800

            Step 6c: Update Booking total
              booking.TotalAmount = 800
              await _context.SaveChangesAsync()

            Step 6d: COMMIT TRANSACTION
              await transaction.CommitAsync()
              // All changes persisted to database atomically

            Step 6e: Return success response
              return Json(new {
                  success = true,
                  message = "Booking confirmed successfully!",
                  bookingId = 42
              });
        }
        catch (Exception ex)
        {
            // On any error:
            await transaction.RollbackAsync()
            // All changes since BeginTransactionAsync() are rolled back

            return Json(new {
                success = false,
                message = "Booking failed. Please try again.",
                errorDetails = ex.Message
            });
        }
    }

KEY SAFETY & SECURITY FEATURES IN FLOW
═════════════════════════════════════════════════════════════════════════════════

1. RACE CONDITION PREVENTION:
   ├─ Issue: Two users book same seat simultaneously
   ├─ Solution 1: Availability check before transaction
   │              └─ Code checks BookedSeats before creating new booking
   ├─ Solution 2: Database-level protection
   │              └─ UNIQUE INDEX on BookedSeat(ShowId, SeatNumber)
   │              └─ If duplicate attempted, SQL constraint violation → rollback
   └─ Solution 3: Transaction atomicity
                 └─ All BookedSeats created in same transaction
                 └─ Either all succeed or all fail

2. CSRF PROTECTION (Cross-Site Request Forgery):
   ├─ Issue: Attacker tricks user into booking from malicious site
   ├─ Solution: Antiforgery token
   │   ├─ @Html.AntiForgeryToken() in form → hidden input field
   │   ├─ jQuery reads token and includes in AJAX header
   │   ├─ [ValidateAntiForgeryToken] on server validates token
   │   └─ Request without valid token → 400 Bad Request

3. USER AUTHORIZATION:
   ├─ Issue: Unauthorized users accessing booking endpoints
   ├─ Solution 1: [Authorize] on controller
   │              └─ Unauthenticated requests rejected
   ├─ Solution 2: Ownership verification
   │              └─ Confirmation action checks: booking.UserId == currentUserId
   │              └─ If mismatch → 403 Forbid
   └─ Solution 3: GetSeatStatus is [AllowAnonymous]
                 └─ Safe to expose (no personal data, no write access)

4. TRANSACTION SAFETY:
   ├─ Issue: Partial booking if error occurs during seat creation
   ├─ Solution: Database transaction (ACID guarantee)
   │   ├─ Atomicity: All BookedSeats created or none
   │   ├─ Consistency: Booking.TotalAmount matches actual seats × prices
   │   ├─ Isolation: Other users don't see incomplete booking
   │   └─ Durability: Once committed, changes survive failures
   └─ If error in transaction:
       └─ transaction.RollbackAsync() cancels all changes
       └─ No partial booking in database

5. INPUT VALIDATION:
   ├─ Server validates: ShowId > 0, SeatNumbers not empty
   ├─ Server validates: Show exists and is active
   ├─ Server validates: Show start time is in future
   └─ Server rejects invalid requests with 400 Bad Request

6. LOGGING & AUDITING:
   ├─ All actions logged: _logger.LogInformation(...)
   ├─ Suspicious access logged: _logger.LogWarning(...)
   ├─ Errors logged: _logger.LogError(...)
   └─ Useful for debugging and security audits

SUMMARY
═════════════════════════════════════════════════════════════════════════════════

Complete flow from user browsing movies to confirmed booking:

  1. HOME → View movie grid with MovieCardViewModel
  2. DETAILS → View showtimes with nested ShowListItemViewModel
  3. SELECT SEATS → Interactive seat grid with jQuery + AJAX
     └─ Real-time availability checks ($.getJSON)
     └─ Dynamic price calculation
  4. RESERVE → AJAX POST with database transaction
     └─ Atomicity: All-or-nothing
     └─ Safety: Race condition prevention
     └─ Security: Antiforgery token, user ownership
  5. CONFIRMATION → Display booking with all details
     └─ Ownership verified (Forbid if not owner)
     └─ Print-friendly layout

All endpoints protected appropriately:
  ✓ [Authorize] where authentication required
  ✓ [AllowAnonymous] where needed for UX (GetSeatStatus)
  ✓ [ValidateAntiForgeryToken] for all mutating operations (ReserveSeats)
  ✓ Ownership checks for sensitive data (Confirmation)
