 * 1. BookedSeat Entity
 *    BEFORE: ShowId was a simple FK property

 *    WHY:    Explicit declaration ensures EF Core treats it as a mandatory FK

 * 2. Booking Entity
 *    BEFORE: Status was string, values like "Pending", "Confirmed", "Cancelled"
 *    AFTER:  Status is enum BookingStatus { Pending, Confirmed, Refunded, Cancelled }
 *    WHY:    Type-safe, prevents invalid status values, easier refactoring
 * 
 * 3. All Main Entities
 *    ADDED:  public DateTime? UpdatedAt { get; set; }
 *    WHY:    Audit trail – track when records were last modified
 *    USED:   Admin reports, debugging, compliance

 * 4. ApplicationUser
 *    ADDED:  public override string? PhoneNumber { get; set; }
 *    ADDED:  public DateTime CreatedAt { get; set; }
 *    ADDED:  public DateTime? UpdatedAt { get; set; }
 *    WHY:    Explicit phone number, user registration tracking

 * ═════════════════════════════════════════════════════════════════════════════════
 * NEW FILE: BookingStatus.cs ENUM
 * ═════════════════════════════════════════════════════════════════════════════════
 * 
 *     public enum BookingStatus
 *     {
 *         Pending = 0,      // Initial state after user books
 *         Confirmed = 1,    // After successful payment
 *         Refunded = 2,     // After refund request approved
 *         Cancelled = 3     // After user/admin cancellation
 *     }
 * 
 * Usage in Booking entity:
 *     public BookingStatus Status { get; set; } = BookingStatus.Pending;
 * 
 * In DbContext fluent API:
 *     entity.Property(e => e.Status)
 *         .HasConversion<int>()
 *         .HasDefaultValue(BookingStatus.Pending);
 * 
 * This stores the enum as int in the database (0, 1, 2, 3).
 * 
 * ═════════════════════════════════════════════════════════════════════════════════
 * HOW DBCONTEXT USES THESE MODELS
 * ═════════════════════════════════════════════════════════════════════════════════
 * 
 * 1. DbSet Properties (one per entity):
 * 
 *    public DbSet<Movie> Movies { get; set; }
 *    public DbSet<Theatre> Theatres { get; set; }
 *    public DbSet<Screen> Screens { get; set; }
 *    public DbSet<Show> Shows { get; set; }
 *    public DbSet<SeatPrice> SeatPrices { get; set; }
 *    public DbSet<Booking> Bookings { get; set; }
 *    public DbSet<BookedSeat> BookedSeats { get; set; }
 * 
 *    These enable querying: db.Movies.ToListAsync(), db.Bookings.FindAsync(id), etc.
 * 
 * 2. OnModelCreating Configurations:
 * 
 *    • Validates annotations on entities (Required, StringLength, Range, etc.)
 *    • Defines relationships using fluent API
 *    • Sets up indexes for performance and uniqueness
 *    • Configures column types (decimal(10,2), nvarchar(max), etc.)
 *    • Seeds initial data (Admin role, Admin user)
 * 
 * 3. Example: BookedSeat Configuration in DbContext
 * 
 *    builder.Entity<BookedSeat>(entity =>
 *    {
 *        entity.HasKey(e => e.Id);  // PK
 *        
 *        entity.Property(e => e.ShowId).IsRequired();
 *        entity.Property(e => e.SeatNumber).IsRequired().HasMaxLength(10);
 *        entity.Property(e => e.Price).HasColumnType("decimal(10, 2)");
 *        
 *        // Relationships
 *        entity.HasOne(bs => bs.Booking)
 *            .WithMany(b => b.BookedSeats)
 *            .HasForeignKey(bs => bs.BookingId)
 *            .OnDelete(DeleteBehavior.Cascade);
 *        
 *        entity.HasOne(bs => bs.Show)
 *            .WithMany(s => s.BookedSeats)
 *            .HasForeignKey(bs => bs.ShowId)
 *            .OnDelete(DeleteBehavior.Cascade);
 *        
 *        // *** UNIQUE INDEX FOR DOUBLE-BOOKING PREVENTION ***
 *        entity.HasIndex(e => new { e.ShowId, e.SeatNumber })
 *            .IsUnique()
 *            .HasName("UX_BookedSeat_ShowId_SeatNumber");
 *    });
 * 
 * ═════════════════════════════════════════════════════════════════════════════════
 * DATA FLOW: BOOKING A SEAT (HOW MODELS WORK TOGETHER)
 * ═════════════════════════════════════════════════════════════════════════════════
 * 
 * 1. USER selects seats on UI (JavaScript checks against booked seats)
 * 
 * 2. JAVASCRIPT AJAX calls:
 *    GET /api/booking/seat-status/5
 *    → Returns list of booked seats for Show #5
 * 
 * 3. CONTROLLER queries:
 *    var bookedSeats = await db.BookedSeats
 *        .Where(bs => bs.ShowId == 5 && bs.IsActive)
 *        .ToListAsync();
 *    
 *    Returns: List<BookedSeat> with SeatNumber, Category, Price
 * 
 * 4. USER selects seats and clicks "Book Now"
 * 
 * 5. JAVASCRIPT AJAX calls:
 *    POST /api/booking/reserve-seats
 *    Body: { showId: 5, selectedSeats: ["A1", "A2"], totalAmount: 600 }
 * 
 * 6. CONTROLLER TRANSACTION:
 *    
 *    using (var transaction = db.Database.BeginTransaction())
 *    {
 *        try
 *        {
 *            // Create Booking
 *            var booking = new Booking
 *            {
 *                UserId = userId,
 *                ShowId = 5,
 *                BookingDateTime = DateTime.UtcNow,
 *                TotalAmount = 600,
 *                Status = BookingStatus.Pending,
 *                CreatedAt = DateTime.UtcNow
 *            };
 *            db.Bookings.Add(booking);
 *            await db.SaveChangesAsync();
 *            
 *            // Create BookedSeats
 *            foreach (var seatNumber in selectedSeats)
 *            {
 *                var bookedSeat = new BookedSeat
 *                {
 *                    BookingId = booking.Id,
 *                    ShowId = 5,
 *                    SeatNumber = seatNumber,
 *                    Category = "Silver",  // from SeatPrice lookup
 *                    Price = 300,
 *                    CreatedAt = DateTime.UtcNow
 *                };
 *                db.BookedSeats.Add(bookedSeat);
 *            }
 *            
 *            await db.SaveChangesAsync();
 *            await transaction.CommitAsync();
 *            
 *            // Success response
 *            return Ok(new { bookingId = booking.Id });
 *        }
 *        catch (DbUpdateException ex)
 *        {
 *            // Unique index violation: seat already booked
 *            await transaction.RollbackAsync();
 *            return Conflict(new { error = "Seat(s) already booked. Please refresh and try again." });
 *        }
 *    }

 * ═════════════════════════════════════════════════════════════════════════════════
 * MIGRATION COMMAND SEQUENCE
 * ═════════════════════════════════════════════════════════════════════════════════
 * 
 * 1. Add NuGet packages (if not already done):
 *    dotnet add package Microsoft.EntityFrameworkCore.SqlServer
 *    dotnet add package Microsoft.EntityFrameworkCore.Tools
 *    dotnet add package Microsoft.EntityFrameworkCore.Design
 *    dotnet add package Microsoft.AspNetCore.Identity.EntityFrameworkCore
 * 
 * 2. Create initial migration:
 *    dotnet ef migrations add InitialCreate
 * 
 *    This creates Migrations/[timestamp]_InitialCreate.cs with:
 *    - CreateTable statements for all entities
 *    - Unique index definitions
 *    - Foreign key relationships
 *    - Seed data (Admin role, Admin user)
 * 
 * 3. Apply migration to database:
 *    dotnet ef database update
 * 
 *    This executes the migration script against SQL Server.
 *    Database is created if it doesn't exist.
 *    Seed data is inserted.
 * 
 * ═════════════════════════════════════════════════════════════════════════════════
 * IMPORTANT: SOFT DELETE PATTERN
 * ═════════════════════════════════════════════════════════════════════════════════
 * 
 * All entities have IsActive boolean field (default = true).
 * 
 * QUERIES MUST FILTER MANUALLY:
 * 
 *    ✓ CORRECT:
 *    var activeMovies = await db.Movies
 *        .Where(m => m.IsActive)
 *        .ToListAsync();
 * 
 *    ✗ WRONG (returns deleted items):
 *    var allMovies = await db.Movies.ToListAsync();
 * 
 * FUTURE: Consider implementing a Query Filter in OnModelCreating:
 * 
 *    builder.Entity<Movie>()
 *        .HasQueryFilter(m => m.IsActive);
 * 
 *    This automatically filters all queries (optional, but recommended for large apps).

 * ═════════════════════════════════════════════════════════════════════════════════
 * QUICK REFERENCE: FIELD SUMMARY BY ENTITY
 * ═════════════════════════════════════════════════════════════════════════════════
 * 
 * ApplicationUser
 *    PK: Id (GUID, inherited from IdentityUser)
 *    Fields: FirstName, LastName, PhoneNumber, IsActive, CreatedAt, UpdatedAt
 *    FK: None (owns Bookings)
 * 
 * Movie
 *    PK: Id (int)
 *    Fields: Title, Description, DurationMinutes, ReleaseDate, PosterUrl, TrailerUrl, 
 *            IsActive, CreatedAt, UpdatedAt
 *    FK: None (owns Shows)
 * 
 * Theatre
 *    PK: Id (int)
 *    Fields: Name, City, Address, IsActive, CreatedAt, UpdatedAt
 *    FK: None (owns Screens)
 * 
 * Screen
 *    PK: Id (int)
 *    Fields: Name, TotalSeats, LayoutJson, IsActive, CreatedAt, UpdatedAt
 *    FK: TheatreId (→ Theatre)
 *    Owns: Shows
 * 
 * Show
 *    PK: Id (int)
 *    Fields: StartDateTime, EndDateTime, IsActive, CreatedAt, UpdatedAt
 *    FK: MovieId (→ Movie), ScreenId (→ Screen)
 *    Owns: SeatPrices, Bookings, BookedSeats
 * 
 * SeatPrice
 *    PK: Id (int)
 *    Fields: Category, Price (decimal), CreatedAt, UpdatedAt
 *    FK: ShowId (→ Show)
 * 
 * Booking
 *    PK: Id (int)
 *    Fields: BookingDateTime, TotalAmount (decimal), Status (enum), IsActive, Notes, 
 *            CreatedAt, UpdatedAt
 *    FK: UserId (→ ApplicationUser), ShowId (→ Show)
 *    Owns: BookedSeats
 * 
 * BookedSeat
 *    PK: Id (int)
 *    Fields: SeatNumber, Category, Price (decimal), IsActive, CreatedAt, UpdatedAt
 *    FK: BookingId (→ Booking), ShowId (→ Show)
 *    UNIQUE: (ShowId, SeatNumber) ← PREVENTS DOUBLE-BOOKING
